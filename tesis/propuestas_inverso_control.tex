\section*{0. Diccionario: $\varphi$, $N$ y $n$ no son lo mismo}

Tres cantidades distintas. Mezclarlas es el error de la
versión anterior.

\begin{itemize}
\item $\varphi$: \textbf{fracción de volumen} ocupada por gas
  (burbujas). Adimensional, $\varphi\in[0,1]$. El fundido ocupa
  $1-\varphi$. Es incógnita de campo: en 1D, $\varphi(z)$; en
  2D, $\varphi(r,z)$. Cambia si las burbujas \emph{crecen}
  (baja $P$, exsuelve agua), aunque el \emph{número} de
  burbujas no cambie.
\item $N$: \textbf{número de burbujas} por unidad de volumen
  de fundido. Unidades $\mathrm{m^{-3}}$. No es un volumen.
  Cambia por coalescencia (se fusionan $\Rightarrow N$ baja)
  o nucleación. Relación geométrica, no evolutiva:
  \begin{equation}
  \label{eq:rb}
  \varphi
  =N\cdot\Bigl(\tfrac43\pi r_b^3\Bigr)\cdot(1-\varphi),
  \qquad
  r_b
  =\Biggl(\frac{\varphi}{\tfrac43\pi N(1-\varphi)}\Biggr)^{1/3}.
  \end{equation}
  $r_b$ es el radio de una burbuja. Mismo $\varphi$, distinto
  $N$ $\Rightarrow$ pocas burbujas grandes o muchas chicas.
  Eso cambia el arrastre $F_{mg}$ y la viscosidad.
\item $n$: \textbf{fracción másica} de agua ya exsuelta (gas).
  Sale de Henry, $n=n(P,\xi)$. No es $\varphi$: una es masa,
  la otra volumen. Con $u_m=u_g$ (sin deslizamiento, p.\,ej.\
  en la base),
  \begin{equation}
  \label{eq:phi-noslip}
  \varphi
  =\Biggl(1+\frac{P}{n\,\Rv T}\frac{1-n}{\rho_m}\Biggr)^{-1}.
  \end{equation}
  Eso \textbf{no} es la ecuación de evolución de $\varphi$.
  Vale solo si $u_m=u_g$. A lo largo del conducto hay slip
  ($u_g\neq u_m$) y $\varphi$ evoluciona con masa + momentum.
\end{itemize}

Las otras: $P$ presión; $\xi$ fracción de cristales en el
fundido; $u_m$, $u_g$ velocidades verticales del fundido y del
gas; $\rho_m$ constante; $\rho_g=P/(\Rv T)$ (EOS del gas,
\textbf{no} la ecuación de $P$).

\section*{Parte I. Estacionario 1D, por leyes}

Dominio $z\in[H,0]$ ($H<0$ es la base). Caso Calbuco:
$H=-7000\,\mathrm{m}$, $R=16\,\mathrm{m}$.
El estado es $\mathbf{y}=(P,\varphi,N,\xi,u_m,u_g)(z)$.
$q$ (caudal másico por unidad de área) es constante; el
$\mathrm{MER}$ es $\pi R^2 q$. $q$ no es dato: sale del tiro.

% ----- MASA -----
\subsection*{I.A Conservación de masa}

\textbf{Variables principales:} $\varphi$, $u_m$, $u_g$, y el
escalar $q$. Entra $n=n(P,\xi)$ como cierre.

Estacionario, área constante $\Rightarrow$ el flujo másico no
cambia con $z$:
\begin{equation}
\label{eq:q}
q
=\rho_m(1-\varphi)\,u_m
+\rho_g\,\varphi\,u_g
=\mathrm{const}.
\end{equation}
Se identifica $n$ de Henry con la fracción del flujo que va
en el gas,
\begin{equation}
\label{eq:n-flux}
n
=\frac{\rho_g\,\varphi\,u_g}{q},
\qquad
1-n
=\frac{\rho_m(1-\varphi)\,u_m}{q}.
\end{equation}
De ahí se \textbf{despejan las velocidades}, no $\varphi$:
\begin{equation}
\label{eq:vel}
u_m
=\frac{q\,(1-n)}{(1-\varphi)\,\rho_m},
\qquad
u_g
=\frac{q\,n}{\varphi\,\rho_g}.
\end{equation}
$u_m$ y $u_g$ son algebraicas. $\varphi$ queda libre: la
cierran el movimiento (I.B) y, en la base,~\eqref{eq:phi-noslip}
porque ahí $u_m=u_g=v_{\mathrm{in}}$.

\textbf{Bordes de masa.}
Base: si $n=0$, $\varphi(H)=0$ y $u_m=u_g=v_{\mathrm{in}}$,
$q=v_{\mathrm{in}}\rho_m$. Si $n>0$, $\varphi(H)$ por
\eqref{eq:phi-noslip} y $q=v_{\mathrm{in}}\rho_{t,i}$.
Boca: no se impone un valor de $u_m$; $\varphi(0)$ entra en
el cierre efusivo/explosivo (I.F).

% ----- MOVIMIENTO / P -----
\subsection*{I.B Conservación de momentum, y la ecuación para $P$}

\textbf{Variables principales:} $P$, $u_m$, $u_g$.
Entran $\varphi$ (fracción de cada fase), las fuerzas y $\mu$.

Estacionario $\Rightarrow$ no hay $\p/\p t$; sí está
$u\,\dd u/\dd z$:
\begin{align}
\label{eq:mom-1d}
\rho_m(1-\varphi)\,u_m\frac{\dd u_m}{\dd z}
&=-(1-\varphi)\frac{\dd P}{\dd z}
-\rho_m(1-\varphi)\,g
+F_{mg}-F_{mw},
\\
\rho_g\varphi\,u_g\frac{\dd u_g}{\dd z}
&=-\varphi\frac{\dd P}{\dd z}
-\rho_g\varphi\,g
-F_{mg}-F_{gw}.
\end{align}
$F_{mw}=c_g\mu u_m/R^2$ (pared--fundido), $F_{gw}=0$ antes de
fragmentar, $F_{mg}$ arrastre entre fases.

Sustituyendo~\eqref{eq:vel} en~\eqref{eq:mom-1d} queda un
$2\times 2$ lineal. \textbf{Ecuación para la presión} (y la
de $\varphi$ a lo largo):
\begin{equation}
\label{eq:P-1d}
\frac{\dd P}{\dd z}
=\frac{-e\,a+f\,c}{ad-bc}
=:\mathcal{P}(P,\varphi,N,\xi;q),
\qquad
\frac{\dd\varphi}{\dd z}
=\frac{-e\,b+d\,f}{ad-bc}
=:\Phi(P,\varphi,N,\xi;q),
\end{equation}
\begin{align*}
a&=\rho_g u_g^2, &
b&=\varphi-\frac{u_g^2\varphi}{\Rv T}+\frac{\p n}{\p P}\,q\,u_g,\\
c&=\rho_m u_m^2, &
d&=\frac{\p n}{\p P}\,q\,u_m-(1-\varphi),\\
e&=-\rho_m(1-\varphi)\,g+F_{mg}-F_{mw}, &
f&=\rho_g\varphi\,g+F_{mg}+F_{gw}.
\end{align*}
$\rho_g=P/(\Rv T)$ es la EOS del gas: da $\rho_g$ dado $P$,
no determina $P$. Quien determina $P$ es~\eqref{eq:P-1d}.

\textbf{Bordes de $P$ y $\varphi$.}
Base: $P(H)=P_{\mathrm{lit}}+\Delta P=\rho_{\mathrm{crust}}g|H|+\Delta P$.
Boca, efusivo: $P(0)=P_{\mathrm{atm}}$ y
$\varphi(0)\le\varphi_{\mathrm{crit}}$.
Boca, explosivo: $P(0)=P_{\mathrm{atm}}$ \textbf{o}
$u_g(0)\approx c_s=\sqrt{\Rv T}$ (flujo sónico, no onda de
choque), y $\varphi(0)\ge\varphi_{\mathrm{crit}}$.

% ----- N -----
\subsection*{I.C Transporte de $N$ (número de burbujas)}

\textbf{Variable principal:} $N$. Entran $\varphi$ y $\mu$
vía $r_b$ y la coalescencia.

\begin{equation}
\label{eq:N-1d}
\frac{\dd N}{\dd z}
=\frac{1}{u_m}\Gamma_N(P,\varphi,N,\xi).
\end{equation}
$\Gamma_N\le 0$ (Slezin): coalescencia. $\Gamma_N=0$ si
$r_b\ge R/2$ o $\varphi\ge\varphi_{\mathrm{crit}}$. Tras
fragmentar, $N$ se congela.

\textbf{Bordes.} Base: $N(H)=10^8\,\mathrm{m^{-3}}$ si no hay
exsolución; si hay, Toramaru
$\log_{10}N=1.5\log_{10}\dot P+5$,
$\dot P=-(\dd P/\dd z)\,v_{\mathrm{in}}$.
Boca: no se impone $N(0)$ (llega desde abajo).

% ----- XI -----
\subsection*{I.D Transporte de $\xi$ (cristales)}

\textbf{Variable principal:} $\xi$. Entra $P$ (equilibrio).

\begin{equation}
\label{eq:xi-1d}
\frac{\dd\xi}{\dd z}
=\frac{1}{u_m}\Gamma_\xi(P,\xi),
\qquad
\Gamma_\xi
=\max\Bigl(0,\;
(\xi_{\max}-\xi_0)\,f_2 f_3/\tau_c\Bigr).
\end{equation}
$f_2,f_3$ interpolan hacia $\xi_{\mathrm{eq}}(P)$. Tras
fragmentar, $\xi$ se congela.

\textbf{Bordes.} Base: $\xi(H)=\xi_0$. Boca: no se impone
$\xi(0)$.

% ----- CIERRES -----
\subsection*{I.E Cierres que entran en las leyes}

\begin{equation}
\label{eq:henry}
n(P,\xi)
=\max\Biggl(0,\;
\frac{(1-\xi_0)c_0-(1-\xi)\,C_1 P^{\beta}}{1-C_1 P^{\beta}}\Biggr).
\end{equation}
$r_b$ por~\eqref{eq:rb}. $F_{mg}=F_{mg}(\varphi,u_g-u_m,r_b)$
cambia al cruzar $\varphi_1,\varphi_2,\varphi_{\mathrm{crit}}$.
$\mu=\mu(P,\varphi,\xi)$.

% ----- ACOPLAMIENTOS 1D -----
\subsection*{I.F Acoplamientos (1D)}

\begin{center}
\small
\begin{tabular}{@{}lll@{}}
\toprule
bloque & necesita de & porque \\
\midrule
masa (vel.) & $P,\xi,\varphi,q$ & $n(P,\xi)$, $\rho_g(P)$,~\eqref{eq:vel} \\
$P$ y $\varphi$ (mom.) & $u_m,u_g,N,\xi$ & inercia, $F_{mg}(r_b)$, $\mu$, $\p n/\p P$ \\
$N$ & $\varphi,u_m,P,\xi$ & $\Gamma_N$, $r_b(\varphi,N)$ \\
$\xi$ & $P,u_m$ & $\Gamma_\xi(P)$ y Henry (vuelve a $n$) \\
tiro $q$ & $P(0)$ o $u_g(0)$ & cierra el sistema \\
\bottomrule
\end{tabular}
\end{center}

Nada se resuelve solo. El lazo corto es
$P\leftrightarrow n\leftrightarrow u_m,u_g\leftrightarrow$ momentum
$\leftrightarrow P$. $N$ y $\xi$ entran más lento (fuerzas y
Henry). $v_{\mathrm{in}}$ (luego $q$) se busca para cumplir
la boca.

\section*{Parte II. El mismo orden en 2D}

Cilindro $\Omega=[0,R]\times[H,0]$. Ahora
$P,\varphi,N,\xi,u_m,u_g,u_{m,r},u_{g,r}$ dependen de $(r,z)$.
Misma plantilla: masa / movimiento+$P$ / $N$ / $\xi$.

% ----- MASA 2D -----
\subsection*{II.A Conservación de masa}

\textbf{Variables principales:} $\varphi$, $\mathbf{u}_m$,
$\mathbf{u}_g$. $\Gamma$ es la tasa de exsolución
(fundido $\to$ gas).

\begin{align}
\label{eq:masa-m}
\frac{1}{r}\frac{\p}{\p r}\bigl(r\,\rho_m(1-\varphi)\,u_{m,r}\bigr)
+\frac{\p}{\p z}\bigl(\rho_m(1-\varphi)\,u_m\bigr)
&=-\Gamma,
\\
\label{eq:masa-g}
\frac{1}{r}\frac{\p}{\p r}\bigl(r\,\rho_g\varphi\,u_{g,r}\bigr)
+\frac{\p}{\p z}\bigl(\rho_g\varphi\,u_g\bigr)
&=\Gamma.
\end{align}
El caudal total es constante en $z$:
\begin{equation}
\label{eq:Q}
Q
=\int_0^R 2\pi r\bigl[
\rho_m(1-\varphi)\,u_m+\rho_g\,\varphi\,u_g
\bigr]\,\dd r
=\mathrm{const}.
\end{equation}
\textbf{Esto} es la ecuación de $\varphi$ en 2D (junto con el
momentum). No se usa~\eqref{eq:phi-noslip} dentro del
conducto: esa fórmula es 1D de \emph{base}, cuando
$u_m=u_g$. En 2D, $n=n(P,\xi)$ sigue siendo Henry; si se
identifica con la fracción local del flujo,
\begin{equation}
\label{eq:n-loc}
n
=\frac{\rho_g\,\varphi\,u_g}
{\rho_m(1-\varphi)\,u_m+\rho_g\,\varphi\,u_g},
\end{equation}
eso cierra \textbf{velocidades} (como~\eqref{eq:vel}), no
impone $\varphi(r,z)$.

\textbf{Bordes de masa.}
Eje: $u_{m,r}=u_{g,r}=0$.
Pared: $u_{m,r}=u_{g,r}=0$ (impermeable).
Base: $\varphi(r,H)=0$ si $n=0$; si no,~\eqref{eq:phi-noslip}
con $P(r,H)$ (ahí se toma $u_m=u_g$).
Boca: $\varphi$ en el cierre efusivo/explosivo, igual que I.B.

% ----- MOVIMIENTO / P 2D -----
\subsection*{II.B Momentum y la ecuación para $P$}

\textbf{Variables principales:} $P$, $\mathbf{u}_m$,
$\mathbf{u}_g$.

Estacionario: otra vez sin $\p\mathbf{u}/\p t$, con
$(\mathbf{u}\cdot\nabla)\mathbf{u}$. Eje $z$:
\begin{align}
\label{eq:mom-m-z}
\rho_m(1-\varphi)\Biggl(
u_{m,r}\frac{\p u_m}{\p r}+u_m\frac{\p u_m}{\p z}
\Biggr)
&=
-(1-\varphi)\frac{\p P}{\p z}
-\rho_m(1-\varphi)\,g
+F_{mg}
\nonumber\\
&\quad
+\frac{1}{r}\frac{\p}{\p r}\Biggl[
r\,\mu(1-\varphi)\frac{\p u_m}{\p r}
\Biggr]
+\frac{\p}{\p z}\Biggl[
\mu(1-\varphi)\frac{\p u_m}{\p z}
\Biggr],
\\
\label{eq:mom-g-z}
\rho_g\varphi\Biggl(
u_{g,r}\frac{\p u_g}{\p r}+u_g\frac{\p u_g}{\p z}
\Biggr)
&=
-\varphi\frac{\p P}{\p z}
-\rho_g\varphi\,g
-F_{mg}
\nonumber\\
&\quad
+\frac{1}{r}\frac{\p}{\p r}\Biggl[
r\,\mu_g\varphi\frac{\p u_g}{\p r}
\Biggr]
+\frac{\p}{\p z}\Biggl[
\mu_g\varphi\frac{\p u_g}{\p z}
\Biggr].
\end{align}
\textbf{Ecuación para $P$} (las dos componentes de $\nabla P$).
De~\eqref{eq:mom-m-z} se despeja $\p P/\p z$. Del momentum
radial del fundido, $\p P/\p r$:
\begin{equation}
\label{eq:P-2d}
\begin{aligned}
(1-\varphi)\frac{\p P}{\p z}
&=
-\rho_m(1-\varphi)\Biggl(
u_{m,r}\frac{\p u_m}{\p r}+u_m\frac{\p u_m}{\p z}
\Biggr)
-\rho_m(1-\varphi)\,g
+F_{mg}
+\mathrm{div}\,\tau_m^{(z)},
\\[0.25em]
(1-\varphi)\frac{\p P}{\p r}
&=
-\rho_m(1-\varphi)\Biggl(
u_{m,r}\frac{\p u_{m,r}}{\p r}+u_m\frac{\p u_{m,r}}{\p z}
\Biggr)
+\mathrm{div}\,\tau_m^{(r)}.
\end{aligned}
\end{equation}
$P=P(r,z)$. No se impone $P=P(z)$ aquí. La reducción
cuando las componentes radiales se desprecian va más
adelante, en~II.E. La EOS $\rho_g=P/(\Rv T)$ sigue sin
ser la ecuación de $P$.

Al integrar el viscoseo radial,
\begin{equation}
\label{eq:Fmw}
F_{mw}
=-\frac{2}{R}\Biggl[\mu(1-\varphi)\frac{\p u_m}{\p r}\Biggr]_{r=R},
\quad
F_{gw}
=-\frac{2}{R}\Biggl[\mu_g\varphi\frac{\p u_g}{\p r}\Biggr]_{r=R}.
\end{equation}
En el PDE no son cuerpos: el no-deslizamiento está en $r=R$.

\textbf{Bordes de movimiento y de $P$.}
Eje: $\p u_m/\p r=\p u_g/\p r=0$, $\p P/\p r=0$.
Pared: $u_m=0$ (no deslizamiento del fundido).
Base: $P(r,H)=P_{\mathrm{lit}}+\Delta P$.
Boca: $P(r,0)=P_{\mathrm{atm}}$, o explosivo
$u_g\approx c_s$.

% ----- N, XI 2D -----
\subsection*{II.C Transporte de $N$ y de $\xi$ en $(r,z)$}

\textbf{Variables principales:} $N(r,z)$ y $\xi(r,z)$.
Mismas $\Gamma_N$, $\Gamma_\xi$ del 1D. El fundido también
se mueve en $r$, así que no son planas en el radio:
\begin{align}
\label{eq:Nxi-2d}
u_{m,r}\frac{\p N}{\p r}+u_m\frac{\p N}{\p z}
&=\Gamma_N(P,\varphi,N,\xi),
\\
u_{m,r}\frac{\p \xi}{\p r}+u_m\frac{\p \xi}{\p z}
&=\Gamma_\xi(P,\xi).
\end{align}

\textbf{Bordes.}
Eje: $\p N/\p r=\p \xi/\p r=0$.
Pared: no se imponen (las arrastra el fundido; $u_m=0$).
Base: $N(r,H)=N_0$, $\xi(r,H)=\xi_0$.
Boca: no se imponen.

% ----- ACOPLAMIENTOS 2D -----
\subsection*{II.D Acoplamientos (2D)}

Los mismos del 1D, más el radio:

\begin{itemize}
\item \textbf{Masa $\leftrightarrow$ movimiento.}
  $\varphi$ y $\mathbf{u}$ están en las dos. $P$ entra a la
  masa por $\rho_g(P)$ y por $n(P,\xi)$. El movimiento
  devuelve $\nabla P$ y el perfil $u_m(r)$ (luego $Q$).
\item \textbf{$P\leftrightarrow n\leftrightarrow\mathbf{u}$.}
  Henry $n(P,\xi)$ cierra~\eqref{eq:n-loc}; las velocidades
  vuelven a~\eqref{eq:P-2d}.
\item \textbf{$N\leftrightarrow\varphi\leftrightarrow F_{mg},\mu$.}
  $r_b(\varphi,N)$ cambia el arrastre y la viscosidad, que
  cambian $P$ y $\varphi$.
\item \textbf{$\xi\leftrightarrow n,\mu,\Gamma_\xi$.}
  $\xi$ entra en Henry y en $\mu$; $P$ entra en
  $\Gamma_\xi$.
\item \textbf{El tiro $Q$.} Se busca $v_{\mathrm{in}}$ para
  cumplir la boca. En 2D, $u_m(r)$ y
  $\varphi,N,\xi$ no planos cambian $Q$: el $\Delta P$
  recuperado no es el del 1D.
\end{itemize}

% ----- REDUCCIÓN -----
\subsection*{II.E Reducción: componentes radiales despreciables}

Se impone
\[
u_{m,r}\equiv 0,
\qquad
u_{g,r}\equiv 0
\]
(no hay flujo ni convección en $r$). Las derivadas
$\p/\p r$ de las \emph{incógnitas} no se tiran: $u_m$,
$\varphi$, $N$ y $\xi$ pueden seguir dependiendo de $r$.
Esto no es el 1D.

\paragraph{Masa.}
Los flujos radiales de~\eqref{eq:masa-m}--\eqref{eq:masa-g}
se anulan. Queda solo la parte en $z$, local en cada $r$:
\begin{align}
\label{eq:masa-red}
\frac{\p}{\p z}\bigl(\rho_m(1-\varphi)\,u_m\bigr)
&=-\Gamma,
\\
\frac{\p}{\p z}\bigl(\rho_g\varphi\,u_g\bigr)
&=\Gamma.
\end{align}
$Q$~\eqref{eq:Q} sigue constante en $z$. En cada $r$ es un
balance 1D, pero acoplado a los otros $r$ por $P$ y por el
perfil $u_m(r)$.

\paragraph{Momentum y $P$.}
En~\eqref{eq:mom-m-z}--\eqref{eq:mom-g-z} cae
$u_{\,\cdot\,,r}\p u/\p r$. Queda la inercia vertical y el
viscoseo (el radial es el que da el perfil de Poiseuille):
\begin{align}
\label{eq:mom-red}
\rho_m(1-\varphi)\,u_m\frac{\p u_m}{\p z}
&=
-(1-\varphi)\frac{\p P}{\p z}
-\rho_m(1-\varphi)\,g
+F_{mg}
+\frac{1}{r}\frac{\p}{\p r}\Biggl[
r\,\mu(1-\varphi)\frac{\p u_m}{\p r}
\Biggr],
\\
\rho_g\varphi\,u_g\frac{\p u_g}{\p z}
&=
-\varphi\frac{\p P}{\p z}
-\rho_g\varphi\,g
-F_{mg}
+\frac{1}{r}\frac{\p}{\p r}\Biggl[
r\,\mu_g\varphi\frac{\p u_g}{\p r}
\Biggr].
\end{align}
(Se omitió $\p/\p z$ del esfuerzo axial: es $O(\varepsilon^2)$
si $\varepsilon=R/|H|\ll 1$.) En el eje $r$, con
$u_{m,r}=0$,~\eqref{eq:P-2d}$_2$ se reduce a
\begin{equation}
\label{eq:P-red}
(1-\varphi)\frac{\p P}{\p r}
=\mathrm{div}\,\tau_m^{(r)}.
\end{equation}
Si ese viscoseo radial también se desprecia (lubricación),
$\p P/\p r=0$ y \textbf{$P=P(z)$}. Entonces~\eqref{eq:P-red}
deja de ser una ecuación para $P(r)$: $P(z)$ sale de
integrar~\eqref{eq:mom-red} (o su promedio en la sección).
$F_{mw}$ y $F_{gw}$ siguen siendo~\eqref{eq:Fmw}.

\paragraph{$N$ y $\xi$.}
Cae el transporte en $r$:
\begin{equation}
\label{eq:Nxi-red}
u_m\frac{\p N}{\p z}
=\Gamma_N(P,\varphi,N,\xi),
\qquad
u_m\frac{\p \xi}{\p z}
=\Gamma_\xi(P,\xi).
\end{equation}
Siguen pudiendo ser $N(r,z)$ y $\xi(r,z)$: $u_m$ y las
$\Gamma$ cambian con $r$.

\paragraph{Qué se perdió y qué no.}
Se perdió el flujo radial y $\p P/\p r$ (si se hace
lubricación). No se perdió la dependencia en $r$ de
$u_m,\varphi,N,\xi$. El 1D de la Parte~I sale de aquí
recién al \emph{promediar} en $r$, imponer
$\varphi=\varphi(z)$, $N=N(z)$, $\xi=\xi(z)$ y reemplazar
el viscoseo radial por $F_{mw}=c_g\mu u_m/R^2$.

\textbf{Bordes que quedan.}
Eje: $\p u_m/\p r=\p u_g/\p r=0$, $\p N/\p r=\p \xi/\p r=0$.
Pared: $u_m=0$. Base y boca: las de $P$, $\varphi$ y el
tiro, iguales que en II.A--II.C. $u_{m,r}=u_{g,r}=0$ ya
no es una condición: es la hipótesis.

Eso es el primer ataque numérico (el 2D que ya corre es
esto, con $\varphi=\varphi(z)$). El sistema de II.A--II.C
es el completo.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section*{Parte III. El inverso}

Directo: se resuelve I, II o la reducción II.E, y se lee
$Q$ (o $q$) en la boca.
\[
\F(\theta)=Q,
\qquad
\theta=\Delta P
\quad\text{o}\quad
\theta=c_0.
\]
Dato $Q^{\mathrm{obs}}$. Primero $\Delta P$ (borde de $P$ en
